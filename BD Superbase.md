-- Création du type ENUM pour les catégories
CREATE TYPE category_slug AS ENUM (
  'electronics', 
  'fashion', 
  'home', 
  'food', 
  'services', 
  'other'
);

-- 1. Table des Utilisateurs (lien aux user Superbase)
CREATE TABLE public.profiles (
  id UUID REFERENCES auth.users ON DELETE CASCADE NOT NULL PRIMARY KEY,
  full_name TEXT,
  avatar_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. Table des Boutiques
CREATE TABLE public.shops (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  owner_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  name TEXT NOT NULL,
  description TEXT,
  whatsapp_number TEXT NOT NULL,
  logo_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. Table des Catégories (utilisant l'ENUM)
CREATE TABLE public.categories (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL, -- Exemple: "Électronique"
  icon_slug TEXT -- pour afficher une icone dynamique
);

-- 4. Table des Produits
CREATE TABLE public.products (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  shop_id UUID REFERENCES public.shops(id) ON DELETE CASCADE NOT NULL,
  category_id BIGINT REFERENCES public.categories(id), -- NOT NULL
  title TEXT NOT NULL,
  description TEXT,
  price NUMERIC NOT NULL,
  image_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 5. Activation de la sécurité (Row Level Security)
alter table profiles enable row level security;
alter table shops enable row level security;
alter table products enable row level security;

-- POLITIQUES DE SÉCURITÉ SIMPLES (Pour commencer)
-- Tout le monde peut voir les produits et les boutiques
create policy "Public shops are viewable by everyone." on shops for select using (true);
create policy "Public products are viewable by everyone." on products for select using (true);

-- Seul le propriétaire peut modifier sa boutique
create policy "Users can insert their own shop." on shops for insert with check (auth.uid() = owner_id);
create policy "Users can update their own shop." on shops for update using (auth.uid() = owner_id);
